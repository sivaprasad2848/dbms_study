Normalization Rules
---------------------------------------------------
->First Normal Form(1NF)

	1)No Empty Field 
			not null/default
	2)Atomicity
			address
			housename,post,place,pin
	3)Key Column	
			It uniquely identify the records

			primary key/unique key

create table contacts(slno int unique,name varchar(100) not null,email 
varchar(100) default 'no-email',mobile bigint not null);

->Second Normal Form(2NF)
	->Table must be in 1NF
	->Avoid Partial Dependency

		Split the table and keep the relationship using PK and FK

Field/Column Alias:- used for changing the selected filed name in display

	select name as customer_name,email as customer_email from contacts;

limiting the records in selection

order by:
	Used for sorting

select *from contacts order by slno;
select *from contacts order by slno desc;

where clause:-Used for spefiying the condition
	Operators in where clause

	1)relational operator(=,<,>,<=,>=,!=)

	select *from contacts where slno>10;
	select *from contacts where slno=10;
	
	2)logical operator( and ,or ,not)

	select *from contacts where slno>=5 and slno<=10;


	3)between
	select *from contacts where slno between 10 and 15;

	4)like
	:- Used for checking the patterns
	select *from contacts where name like "%Manu%";
	select *from contacts where name like "M%";

	5)in:- Used for speciying a group

	select *from contacts where slno in(1,2,3);

	6)Subquery
	
	select *from contacts where slno in(select slno from contacts where
	slno <10);

	7)group by

---------------------------------------------------------
Altering Table Strcuture
--------------------------------------------------------
Adding new column to a table

	alter table <table-name> add <column-name> <datatype>;
	Eg:
	alter table products add stock int;
	
Removing a column from table
	alter table <table-name> drop <column-name>;
	EG:
	alter table products drop stock;

Changing the column-name or datatype
	alter table <table-name> rename <column-name> to <new-column-name>;

	Eg:
	alter table products rename quantity to qty;

	For changing datatype in postgres 

	alter table <table-name> alter column <column-name> type <datatype>;
